/** * ===== TTD LOGISTICS — DRIVE UPLOADER (Apps Script + Drive RIÊNG) ===== * Lưu file logistics vào: / / ... * * CÀI ĐẶT: * 1. Tạo 1 folder Drive mới, vd "TTD Logistics". Mở folder → copy ID trên URL * (drive.google.com/drive/folders/) → dán vào ROOT_ID bên dưới. * 2. script.google.com → Dự án mới → dán toàn bộ file này → Lưu. * 3. Triển khai → Lần triển khai mới → ⚙️ Ứng dụng web * - Thực thi với tư cách: Tôi * - Ai có quyền truy cập: Bất kỳ ai * → Triển khai → Ủy quyền → Cho phép. * 4. Copy URL .../exec → gửi cho Claude để dán vào logistics/index.html. * * ***** QUAN TRỌNG: sau khi dán lại file này phải TRIỂN KHAI PHIÊN BẢN MỚI ***** * (Triển khai → Quản lý các lần triển khai → ✏️ → Phiên bản: Mới → Triển khai) * thì action 'scan' (khôi phục đính kèm) mới hoạt động. URL .../exec giữ nguyên. */ const UPLOAD_SECRET = 'ttd_th_2026'; // giữ nguyên, khớp với logistics/index.html const ROOT_ID = '1kjC4SZreyB1ACW_dzZps9ynINls78G7Y'; // ID folder "TTD Logistics" riêng function doPost(e) { try { const b = JSON.parse(e.postData.contents); if (b.secret !== UPLOAD_SECRET) return out({ ok: false, error: 'Sai khóa bảo mật' }); if (b.action === 'delete') { if (b.fileId) { try { DriveApp.getFileById(b.fileId).setTrashed(true); } catch (_) {} } return out({ ok: true }); } // ----- QUÉT TOÀN BỘ: trả về danh sách thư mục + file để KHÔI PHỤC đính kèm ----- if (b.action === 'scan') { const root = DriveApp.getFolderById(ROOT_ID); const folders = root.getFolders(); const result = []; while (folders.hasNext()) { const f = folders.next(); const files = []; const it = f.getFiles(); while (it.hasNext()) { const file = it.next(); const id = file.getId(); files.push({ fileId: id, name: file.getName(), size: file.getSize(), url: 'https://drive.google.com/file/d/' + id + '/view', previewUrl: 'https://drive.google.com/file/d/' + id + '/preview' }); } result.push({ folder: f.getName(), files: files }); } return out({ ok: true, folders: result }); } // ----- Nén nhiều file thành 1 zip để tải về ----- if (b.action === 'zip') { const files = b.files || []; const blobs = []; files.forEach(function(f){ try { const bl = DriveApp.getFileById(f.fileId).getBlob(); if (f.name) bl.setName(f.name); blobs.push(bl); } catch (_) {} }); if (!blobs.length) return out({ ok: false, error: 'Không có file để nén' }); const zipName = (b.zipName || 'chung_tu') + '.zip'; const zip = Utilities.zip(blobs, zipName); return out({ ok: true, zipBase64: Utilities.base64Encode(zip.getBytes()), zipName: zipName }); } const { itemId, itemName, fileName, mimeType, dataBase64 } = b; if (!itemId || !fileName || !dataBase64) return out({ ok: false, error: 'Thiếu tham số' }); const root = DriveApp.getFolderById(ROOT_ID); const displayName = (itemName ? itemName + ' - ' : '') + itemId; const itemFolder = getOrCreateItem(root, itemId, displayName); const bytes = Utilities.base64Decode(dataBase64); const blob = Utilities.newBlob(bytes, mimeType || 'application/octet-stream', fileName); const file = itemFolder.createFile(blob); try { file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.EDIT); } catch (_) {} const id = file.getId(); return out({ ok: true, fileId: id, name: file.getName(), size: file.getSize(), url: 'https://drive.google.com/file/d/' + id + '/view', previewUrl: 'https://drive.google.com/file/d/' + id + '/preview' }); } catch (err) { return out({ ok: false, error: String(err) }); } } function getOrCreate(parent, name) { const it = parent.getFoldersByName(name); return it.hasNext() ? it.next() : parent.createFolder(name); } function getOrCreateItem(parent, id, displayName) { const all = parent.getFolders(); while (all.hasNext()) { const f = all.next(); if (f.getName() === displayName || f.getName().endsWith(id)) return f; } return parent.createFolder(displayName); } function out(o) { return ContentService.createTextOutput(JSON.stringify(o)).setMimeType(ContentService.MimeType.JSON); } // ===== CHẠY 1 LẦN: mở quyền CHỈNH SỬA cho MỌI file đã upload trước đó ===== function setAllEditable() { var root = DriveApp.getFolderById(ROOT_ID); var count = openEditRecursive_(root); return 'Đã mở quyền sửa cho ' + count + ' file.'; } function openEditRecursive_(folder) { var n = 0; var files = folder.getFiles(); while (files.hasNext()) { var f = files.next(); try { f.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.EDIT); n++; } catch (_) {} } var subs = folder.getFolders(); while (subs.hasNext()) { n += openEditRecursive_(subs.next()); } return n; }
